def acquire_single_instance():
	try:
		os.makedirs(os.path.dirname(LOCK_FILE), exist_ok=True)
		fd = os.open(LOCK_FILE, os.O_CREAT | os.O_RDWR)
	except Exception:
		return None
	try:
		import msvcrt
		msvcrt.locking(fd, msvcrt.LK_NBLCK, 1)
	except Exception:
		return None
	return fd


def port_open(timeout=1.0):
	s = socket.socket()
	s.settimeout(timeout)
	try:
		s.connect((HOST, PORT))
		return True
	except Exception:
		return False
	finally:
		s.close()
